<%| site_name ~= '' and site_name or hostname %> <%|hostname%>
<% if model ~= '' then %><%|model%><% end %><%|release%> <% if #categories > 1 and not hidenav then %>
<% for _, r in ipairs(categories) do %> aria-current="page"<% end %>><%|title(root.nodes[r])%> <% end %>
<% end %>
<% if #tabs > 1 and not hidenav then %>
<% for _, r in ipairs(tabs) do %> aria-current="page"<% end %>><%|title(root.nodes[category].nodes[r])%> <% end %>
<% end %>
